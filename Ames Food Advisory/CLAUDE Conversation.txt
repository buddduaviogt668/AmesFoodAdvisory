def build_includes(items):
    rows = []
    for icon,title,desc in items:
        rows.append(f'<div class="include-item fade-up"><div class="include-icon">{icon}</div><div><div class="include-title">{title}</div><div class="include-desc">{desc}</div></div></div>')
    return '<div class="includes-grid">'+''.join(rows)+'</div>'

def build_who(who_title, who_items):
    cards=''.join(f'<div class="who-card fade-up"><div class="who-card-title">{t}</div><div class="who-card-desc">{d}</div></div>' for t,d in who_items)
    return f'<section class="who-section"><div class="container"><div class="section-eyebrow" style="color:var(--amber-light);">{who_title}</div><h2 class="display-light">This service is <em>right for you</em> if...</h2><div class="who-grid">{cards}</div></div></section>'

def build_steps(steps):
    rows=''.join(f'<div class="step fade-up"><div class="step-num">0{i}</div><div class="step-content"><div class="step-title">{t}</div><div class="step-desc">{d}</div></div></div>' for i,(t,d) in enumerate(steps,1))
    return '<div class="steps">'+rows+'</div>'

def build_faq(faqs):
    items_html=''.join(f'<div class="faq-item"><div class="faq-q">{q}</div><div class="faq-a">{a}</div></div>' for q,a in faqs)
    schema_items=[{"@type":"Question","name":q,"acceptedAnswer":{"@type":"Answer","text":a}} for q,a in faqs]
    schema=json.dumps({"@context":"https://schema.org","@type":"FAQPage","mainEntity":schema_items},indent=2)
    return '<div class="faq-list">'+items_html+'</div>', f'<script type="application/ld+json">{schema}</script>'

def build_related(related):
    cards=''.join(f'<a href="{url}" class="related-card fade-up"><div class="related-num">Service {num}</div><div class="related-title">{title}</div><div class="related-desc">{desc}</div></a>' for url,num,title,desc in related)
    return '<div class="related-grid">'+cards+'</div>'
